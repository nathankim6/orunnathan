# 레슨 데이터 작성 규격 (units/uNN.js 의 lessons[] 한 항목)

원문 지문은 `src/orig/all.json` 에 있다 (`src/orig/raw.tsv` → `src/split.py` → `src/typo.py` 로 만든다).
유닛 파일에서는 `sent: T(n).sent, kor: T(n).kor` 로 그대로 가져다 쓴다 — **한 글자도 고치지 않는다**.

```js
{
 no:"01", key:"translation", accent:"#5B57A6", tint:"#EEEDF8", deep:"#3C3878",
 en:"영문 제목", ko:"우리말 제목",
 goal:"…설명할 수 있다.",                 // 학습 목표 한 줄
 fig:"Figure 1  삽화 설명.",               // 두 칸 띄어쓰기로 앞머리와 설명을 나눈다
 tip:"먼저 생각해 보자. …?",               // 앞머리 + 질문
 sent:[ … ], kor:[ … ],                    // 원문 그대로 (길이 같아야 함)
 bank:[["word","③","뜻"], …6개],           // 원문에 실제로 나오는 낱말 · 원문자는 그 낱말이 처음 쓰인 문장 번호
 defs:[["word","english definition"], …6], // bank 와 같은 순서·같은 낱말. 정의문 60자 이내
 defOrder:[2,0,5,1,4,3],                   // 0–5 의 순열 (영영풀이를 섞는 순서)
 flow:[ ["Stage","영문 내용",정답|null], …5행 ],   // 첫 행만 정답 null(제시), 나머지 4행은 빈칸
                                           // 빈칸 행의 내용에는 (  ①  ) 처럼 원문자 빈칸 표시가 있어야 한다
 flowBogi:"a · b · c · d · e · f",         // 6개, 정답 4개를 포함
 para:[ ["② 원문에서 따온 구", "바꿔 쓴 문장 ______ 포함.", "정답"], …5개 ],
                                           // 앞 원문자 뒤 원문 인용 52자 이내 · 바꿔 쓴 문장 56자 이내
                                           // 정답을 채운 문장이 52자 이내여야 한다(교사용에서 두 줄이 되면 안 됨)
 paraBogi:"…8개 · 로 구분",
 check:[
  {t:"title", q:"이 글의 제목으로 가장 알맞은 것은?", ch:[5개 영문 제목], ans:2},
  {t:"fact",  q:"이 글의 내용과 일치하지 <b>않는</b> 것은?", ch:[5개 우리말], ans:4},
  {t:"write", q:"…우리말 한 문장으로 써 보세요.", ans:"모범 답안"}
 ],
 fl:{
  model:{ n:"⑦", toks:[["말","역할"], …], ko:"뼈대 해석" },   // 토큰을 공백으로 이으면 원문 문장과 같아야 한다
  drill:[ {n:"②", en:"원문 문장 그대로", ans:"S … · △V … · [that] S′ … · △V′ …", ko:"해석"}, …3개 ]
 },
 syn:[ {n:"④", name:"구문 이름", q:"«밑줄부» 를 « » 로 감싼 인용", d:"<b>…</b> 설명", k:"해석"}, …2개 ],
 synd:[ {u:"구문 1", en:"새 예문", k:"해석"}, {u:"구문 2", …}, {u:"구문 1 + 2", …} ],
 wtype:[…5], stype:[…5], why:[[…], …5], src:[[…], …5],
 kb:{ title:"…", lead:"…", items:[["소제목","내용"], …3], ask:"생각해 볼 것 …?" }
}
```

## 지켜야 할 것
- `sent`·`kor` 는 `src/orig/all.json` 그대로. 문장 수가 8~18로 제각각이므로 **원문자 번호가 범위를 넘지 않게** 한다.
- `fl.model.n` 과 `fl.drill[].n` 은 서로 겹치지 않는다. drill 의 `en` 은 지문 문장과 **글자까지 같아야** 한다.
- `flow` 정답 4개는 모두 `flowBogi` 안에, `para` 정답 5개는 모두 `paraBogi` 안에 있어야 한다.
- `wtype` 은 `정답·무관·반대·지엽·배경`, `stype` 은 `일치·반대·과장·혼동·시점` 중에서 고른다.
  `check[0].ans` 위치의 wtype 은 `정답`, `why` 도 `["정답"]`. `check[1].ans` 위치의 stype 은 `일치`가 아니어야 하고 나머지는 전부 `일치`.
- `src[i]` 는 근거 문장 번호(원문자). 오답 위치에는 `["⑫와 어긋남 — 이유"]` 처럼 적는다.
- Knowledge Bank 의 사실은 검증 가능한 것만. 불확실한 통설은 넣지 않는다.
- 지시문 어투는 `~하세요 / ~해 보세요` (`~하시오` 금지).
